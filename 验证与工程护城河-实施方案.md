方案 C：Python 包装脚本（最灵活）
创建 run_backtest_suite.py：
#!/usr/bin/env python3
"""
回测套件：回测 + 监控 + 报告
"""
import subprocess
import sys
from datetime import datetime

def run_backtest(timerange):
    """运行回测"""
    cmd = [
        'docker', 'compose', 'run', '--rm', 'freqtrade',
        'backtesting',
        '--config', 'user_data/config_backtest.json',
        '--strategy', 'WyckoffV7_TrueMethod',
        '--timerange', timerange,
        '--cache', 'none',
        '--export', 'trades'
    ]
    
    print(f"🚀 运行回测: {timerange}")
    result = subprocess.run(cmd)
    return result.returncode == 0

def run_monitoring():
    """运行监控分析"""
    cmd = [
        'docker', 'compose', 'run', '--rm',
        '--entrypoint', 'python', 'freqtrade',
        '/freqtrade/user_data/analyze_backtest_with_monitoring.py'
    ]
    
    print("\n📊 运行监控分析...")
    result = subprocess.run(cmd)
    return result.returncode == 0

def main():
    timerange = sys.argv[1] if len(sys.argv) > 1 else '20230101-20260228'
    
    # 运行回测
    if not run_backtest(timerange):
        print("❌ 回测失败")
        sys.exit(1)
    
    # 运行监控
    if not run_monitoring():
        print("⚠️ 监控分析失败（但回测成功）")
        sys.exit(1)
    
    print("\n✅ 完成！")

if __name__ == '__main__':
    main()

            )

1. 开发新功能 → 用规则开关包裹
2. 默认禁用 → 不影响现有系统
3. 环境变量测试 → 快速 A/B 对比
4. 监控分析 → 评估效果

6. 审计日志 → 记录所有变更

方案 C：Python 调度器（跨平台）
创建 monitoring_scheduler.py：
#!/usr/bin/env python3
"""
监控报告调度器
"""
import schedule
import time
import subprocess
from datetime import datetime
from pathlib import Path

def run_weekly_monitoring():
    """运行每周监控"""
    date_str = datetime.now().strftime('%Y%m%d')
    report_dir = Path(f'monitoring_reports/{date_str}')
    report_dir.mkdir(parents=True, exist_ok=True)
    
    print(f"🚀 开始每周监控: {date_str}")
    
    # 运行回测
    subprocess.run([
        'docker', 'compose', 'run', '--rm', 'freqtrade',
        'backtesting',
        '--config', 'user_data/config_backtest.json',
        '--strategy', 'WyckoffV7_TrueMethod',
        '--timerange', '20230101-20260228',
        '--cache', 'none',
        '--export', 'trades'
    ])
    
    # 生成监控报告
    subprocess.run([
        'docker', 'compose', 'run', '--rm',
        '--entrypoint', 'python', 'freqtrade',
        '/freqtrade/user_data/analyze_backtest_with_monitoring.py'
    ])
    
    # 复制报告
    import shutil
    shutil.copy('user_data/detector_monitoring_report.md', report_dir)
    shutil.copy('user_data/enter_tag_winrate_report.md', report_dir)
    
    print(f"✅ 报告已保存到 {report_dir}")

# 每周日凌晨 2 点运行
schedule.every().sunday.at("02:00").do(run_weekly_monitoring)

print("📅 监控调度器已启动")
print("   每周日凌晨 2:00 运行")

while True:
    schedule.run_pending()
    time.sleep(60)
